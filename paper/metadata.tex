\supervisor{dr Mateusz Buczyński}
\thesisplace{Warsaw, June 2027}

% At most 800 characters with spaces.
\thesisabstract{Banks and regulators rely on one-day Value-at-Risk forecasts. This thesis tests whether
a GARCH model whose conditional variance is produced by a neural network forecasts Value-at-Risk better
than standard GARCH models on the WIG20, PX and BUX indices in 2015--2026. Forecasts are evaluated with
a rolling window and formal backtests.}
